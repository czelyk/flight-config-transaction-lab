cmake_minimum_required(VERSION 3.20)
project(flight_config_transaction_lab VERSION 0.1.0 LANGUAGES C)
if(ZIG_EXECUTABLE)
    # Compiler platform detection may replace the toolchain's archive templates.
    set(CMAKE_C_ARCHIVE_CREATE "<CMAKE_AR> ar qc <TARGET> <LINK_FLAGS> <OBJECTS>")
    set(CMAKE_C_ARCHIVE_APPEND "<CMAKE_AR> ar q <TARGET> <LINK_FLAGS> <OBJECTS>")
    set(CMAKE_C_ARCHIVE_FINISH "")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(littlefs STATIC third_party/littlefs/lfs.c third_party/littlefs/lfs_util.c)
target_include_directories(littlefs PUBLIC third_party/littlefs)
target_compile_definitions(littlefs PUBLIC LFS_NO_MALLOC LFS_NO_DEBUG LFS_NO_WARN LFS_NO_ERROR)
add_library(flight_config STATIC src/fc_format.c src/fc_store.c)
target_include_directories(flight_config PUBLIC include)
target_link_libraries(flight_config PUBLIC littlefs)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(flight_config PRIVATE -Wall -Wextra -Werror)
endif()
if(BUILD_TESTING)
    foreach(test unit campaign)
        add_executable(fc_${test} tests/${test}.c tests/ram_nor.c)
        target_include_directories(fc_${test} PRIVATE tests)
        target_link_libraries(fc_${test} PRIVATE flight_config)
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(fc_${test} PRIVATE -Wall -Wextra -Werror)
        endif()
    endforeach()
    add_test(NAME unit COMMAND fc_unit)
    add_test(NAME sampled_power_cuts COMMAND fc_campaign)
    add_test(NAME complete_prefix_trace COMMAND fc_campaign --full-prefix)
endif()
